 void display(int arr[],int n)
 {
     printf("Elemebnts are:\n");
     for (int i=0; i<n; i++)
     {
         printf("%d\n", *(arr+i));
         
     }
 }
void insertionSort (int arr[], int n)
     {
         int i,j,temp;
         printf("Performing insertion sort...");
         for (int i=0; i<n; i++)
         {
             temp = arr[i];
             j=i-1;
             while((temp<arr[j])&& (j>=0))
             {
                 arr[j+1] = arr[j];
                 j--;
                 
             }
             arr[j+1]=temp;
         }
     }
