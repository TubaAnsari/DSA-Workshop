#include<stdio.h>
#define MAX 10
//bubble
int main()
{
    int a[MAX];
    int i; //outer loop
    int n; //limit
    int j;//inner loop
    int temp;
    
    printf("Enter limits: ");
    scanf("%d", &n);

    printf("Enter elements:\n ");
    for(i=0; i<n; i++)
    {
        scanf("%d",&a[i]);
    }
    
    printf("Entered elements:\n ");
    for(i=0; i<n; i++)
    {
        printf("%d\n",a[i]);
    }
    
    printf("BUBBLE SORT\n ");
    
    //PASS
    for(i=0; i<n-1; i++)
    {
        printf("PASS %d\n",i+1);
        for(j=0; j<n-1; j++)
        {
                if (a[j]> a[j+1])
                {
                    temp=a[j];
                    a[j]=a[j+1];
                    a[j+1]=temp;
                }
            }
    } 
        printf("Sorted\n");
        for(i=0; i<n; i++)
            {
                printf("%d\t", a[i]);
            }
    
}
